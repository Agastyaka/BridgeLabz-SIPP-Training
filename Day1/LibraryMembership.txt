CLASS Member
	ATTRIBUTES : name, id, membershipdate
	CONSTRUCTOR(name, id, membershipdate)
		SET this.name=name
		SET this.id=id
		SET this.membershipdate=membershipdate
	METHOD IsMembershipValid(currentdate)
	IF currentdate - membershipdate <= 365
		RETURN true
	ELSE
		RETURN false
Main
	CREATE Member("any",234,"2024-05-01"
	result = CALL IsMembershipValid("2025-04-20")
	PRINT result
