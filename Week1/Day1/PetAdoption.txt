CLASS Pet
	ATTRIBUTES: name, age
	CONSTRUCTOR(name, age)
		SET this.name = name
		SET this.age = age
	METHOD display()
		PRINT name, age

CLASS Dog EXTENDS Pet
	ATTRIBUTES: breed
	CONSTRUCTOR(name, age, breed)
		CALL super(name, age)
		SET this.breed = breed
	METHOD display()
		CALL super.display()
		PRINT breed

CLASS Cat EXTENDS PET
	Attributes: breed
	CONSTRUCTOR(name, age, breed)
		CALL super(name, age)
		SET this.breed = breed
	METHOD display()
		CALL super.display()
		PRINT breed
CLASS Bird EXTENDS PET
	Attributes: species
	CONSTRUCTOR(name, age, species)
		CALL super(name, age)
		SET this.species = species
	METHOD display()
		CALL super.display()
		PRINT species

MAIN
	CREATE Dog("happy",12,"golden")
	CREATE Cat("frodo",2,"Persian")
	CREATE Bird("par",4,"parrot")
CALL display on each
